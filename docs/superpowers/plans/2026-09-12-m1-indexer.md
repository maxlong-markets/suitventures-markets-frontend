# M1 Indexer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An Envio HyperIndex indexer for Robinhood Chain that discovers every Uniswap v4 pool pairing a memecoin with an allowlisted quote asset, tracks swaps, liquidity, candles, holders, supply and Chainlink USD prices, and serves them over GraphQL for the Markets and token pages.

**Architecture:** One `PoolManager` contract feed does discovery (`Initialize` filtered to pools with exactly one allowlisted side) and activity (`Swap`, `ModifyLiquidity`, `Donate`, gated to tracked pools). Each discovered token's ERC-20 is registered dynamically for `Transfer`; Chainlink proxies are indexed for aggregator rotation and aggregators for `AnswerUpdated`. Everything is stored in quote units; USD conversion happens in the frontend's API layer.

**Tech Stack:** Envio HyperIndex 3.10.0 (TypeScript, ESM, Vitest, `createTestIndexer`), viem, `@uniswap/v3-sdk` + `jsbi` for tick math, Node 22+, npm.

**Spec:** `docs/superpowers/specs/2026-09-12-m1-live-data-design.md` (frontend repo). Decisions D21 to D30 in the project README Decision Log.

## Global Constraints

- Chain id **4663** only. HyperSync is the data source; `ENVIO_API_TOKEN` is required for codegen and sync.
- Addresses: PoolManager `0x8366a39CC670B4001A1121B8F6A443A643e40951`; Doppler hook `0x4e3468951D49f2EEa976eD0D6e75fFCb44a9a544` (origin LONG); Pons MemeHook `0xE5e702641Ea86F4ae6cC3cDaeD2B886f976Be044` (origin Pons); USDG `0x5fc5360d0400a0fd4f2af552add042d716f1d168` (6 decimals, USD pinned at 1).
- Entity ids are **lowercase** addresses / poolIds. Event param addresses arrive checksummed; always `.toLowerCase()` before using one as an id.
- All prices, volumes and TVL are stored in **quote units** (never USD). `tvlToken` / `tvlQuote` / balances / supply are raw `BigInt`; prices and volumes are `Float`.
- Every environment variable is `ENVIO_`-prefixed (`ENVIO_API_TOKEN`, `ENVIO_RPC_URL`, `ENVIO_CANDLES_1M_START_BLOCK`, `ENVIO_CANDLES_5M_START_BLOCK`).
- Candle buckets are created only on trades ("gapless" = a bucket after a gap opens at the previous pool price). 1m and 5m candles are created only from their start blocks; 15m, 1h, 1d over the full range. No pruning.
- Origin is a label from the hook address (or the curated override), never a gate. The Swap event is pre-hook-fee; comment it where amounts are read.
- No `multicall` in scripts or effects: batch reads with `Promise.all` over `readContract` (viem's `http(url, { batch: true })` batches the JSON-RPC calls), because Multicall3 is not confirmed on chain 4663.

```ts
import quotesJson from "../../config/quotes.json" with { type: "json" };

export type QuoteRow = {
  address: string;
  symbol: string;
  name: string;
  decimals: number;
  chainlinkProxy: string | null;
  chainlinkAggregator: string | null;
  chainlinkDecimals: number | null;
};

/** Quote allowlist rows, addresses lowercased. Adding a quote is a row here plus a backfill. */
export const QUOTES: QuoteRow[] = (quotesJson as QuoteRow[]).map((q) => ({
  ...q,
  address: q.address.toLowerCase(),
  chainlinkProxy: q.chainlinkProxy?.toLowerCase() ?? null,
  chainlinkAggregator: q.chainlinkAggregator?.toLowerCase() ?? null,
}));

const BY_ADDRESS = new Map(QUOTES.map((q) => [q.address, q]));

export const QUOTE_ADDRESSES: string[] = QUOTES.map((q) => q.address);

export function isQuote(address: string): boolean {
  return BY_ADDRESS.has(address.toLowerCase());
}

export function quoteRow(address: string): QuoteRow | undefined {
  return BY_ADDRESS.get(address.toLowerCase());
}

/** Quote row for a Chainlink proxy or aggregator address, if any. */
export function quoteForFeed(feedAddress: string): QuoteRow | undefined {
  const a = feedAddress.toLowerCase();
  return QUOTES.find((q) => q.chainlinkProxy === a || q.chainlinkAggregator === a);
}
```

`src/lib/origin.ts`:
```ts
import curated from "../../config/curated-pools.json" with { type: "json" };
import { DOPPLER_HOOK, PONS_MEME_HOOK, ZERO_ADDRESS } from "./constants";

export type Origin = "LONG" | "Pons" | "Uniswap" | "Other" | "SUIT" | "SV";

const CURATED = new Map<string, Origin>(
  Object.entries(curated as Record<string, Origin>).map(([id, o]) => [id.toLowerCase(), o]),
);

/** Origin is a label only: hook address, or the curated override. Never a gate. */
export function originFor(poolId: string, hooks: string): Origin {
  const override = CURATED.get(poolId.toLowerCase());
  if (override) return override;
  const h = hooks.toLowerCase();
  if (h === DOPPLER_HOOK) return "LONG";
  if (h === PONS_MEME_HOOK) return "Pons";
  if (h === ZERO_ADDRESS) return "Uniswap";
  return "Other";
}
```

```ts
import { client } from "./rpc.ts";

const target = Number(process.argv[2]);
if (!Number.isFinite(target)) {
  console.error("usage: npm run block-at -- <unix seconds>");
  process.exit(1);
}

let lo = 0n;
let hi = await client.getBlockNumber();
while (lo < hi) {
  const mid = (lo + hi) / 2n;
  const block = await client.getBlock({ blockNumber: mid });
  if (Number(block.timestamp) < target) lo = mid + 1n;
  else hi = mid;
}
const found = await client.getBlock({ blockNumber: lo });
```

`scripts/build-quotes.ts`:
```ts
import { readFileSync, writeFileSync } from "node:fs";
import { parseAbi } from "viem";
import { client } from "./rpc.ts";

const ERC20 = parseAbi([
  "function symbol() view returns (string)",
  "function name() view returns (string)",
  "function decimals() view returns (uint8)",
]);

// Quotes Pons accepts that are not stock tokens. Add symbols here to exclude them.
const EXCLUDED_SYMBOLS = new Set(["WETH", "ETH", "cbBTC", "TAO", "WYFI", "SKHY", "BULL"]);
const USDG = "0x5fc5360d0400a0fd4f2af552add042d716f1d168";

type Row = {
  address: string; symbol: string; name: string; decimals: number;
  chainlinkProxy: string | null; chainlinkAggregator: string | null; chainlinkDecimals: number | null;
};

const snapshot = JSON.parse(readFileSync("data/pons-graduated-pools-2026-09-10.json", "utf8")) as { quoteToken: string }[];
const existing = new Map((JSON.parse(readFileSync("config/quotes.json", "utf8")) as Row[]).map((r) => [r.address.toLowerCase(), r]));
const addresses = [...new Set(snapshot.map((p) => p.quoteToken.toLowerCase()))];

const rows: Row[] = [];
for (const address of addresses) {
  const target = address as `0x${string}`;
  const [symbol, name, decimals] = await Promise.all([
    client.readContract({ address: target, abi: ERC20, functionName: "symbol" }),
    client.readContract({ address: target, abi: ERC20, functionName: "name" }),
    client.readContract({ address: target, abi: ERC20, functionName: "decimals" }),
  ]);
  if (address !== USDG && EXCLUDED_SYMBOLS.has(symbol)) continue;
  const prev = existing.get(address);
  rows.push({
    address, symbol, name, decimals: Number(decimals),
    chainlinkProxy: prev?.chainlinkProxy ?? null,
    chainlinkAggregator: prev?.chainlinkAggregator ?? null,
    chainlinkDecimals: prev?.chainlinkDecimals ?? null,
```

```ts
import { formatUnits } from "viem";

const Q96 = 2 ** 96;

/**
 * Token price in quote units from a v4 sqrtPriceX96.
 * price of currency0 in currency1 = (sqrtP / 2^96)^2 * 10^(dec0 - dec1); inverted when the token is currency1.
 * Doubles are fine for display (relative error about 1e-16).
 */
export function priceFromSqrtPriceX96(
  sqrtPriceX96: bigint,
  decimals0: number,
  decimals1: number,
  tokenIsCurrency0: boolean,
): number {
  const ratio = Number(sqrtPriceX96) / Q96;
  const price1per0 = ratio * ratio * 10 ** (decimals0 - decimals1);
  return tokenIsCurrency0 ? price1per0 : 1 / price1per0;
}

export function toUnits(raw: bigint, decimals: number): number {
  return Number(formatUnits(raw, decimals));
}

export function absBig(x: bigint): bigint {
  return x < 0n ? -x : x;
}
```

```ts
export type Interval = "m1" | "m5" | "m15" | "h1" | "d1";

export const INTERVALS: Interval[] = ["m1", "m5", "m15", "h1", "d1"];

export const INTERVAL_SECONDS: Record<Interval, number> = {
  m1: 60,
  m5: 300,
  m15: 900,
  h1: 3600,
  d1: 86400,
};

export function bucketStart(timestamp: number, interval: Interval): number {
  const s = INTERVAL_SECONDS[interval];
  return Math.floor(timestamp / s) * s;
}

export function candleId(poolId: string, interval: Interval, bucket: number): string {
  return `${poolId}-${interval}-${bucket}`;
}

function startBlock(name: string): number | undefined {
  const raw = process.env[name];
  if (raw === undefined || raw === "") return undefined;
  const n = Number(raw);
  return Number.isFinite(n) ? n : undefined;
}

/**
 * 1m and 5m candles are created only from their configured start blocks (spec section 5);
 * unset means disabled, so a forgotten variable cannot explode the backfill.
 */
export function candleWindowAllows(interval: Interval, blockNumber: number): boolean {
  if (interval === "m1") {
    const from = startBlock("ENVIO_CANDLES_1M_START_BLOCK");
    return from !== undefined && blockNumber >= from;
  }
  if (interval === "m5") {
    const from = startBlock("ENVIO_CANDLES_5M_START_BLOCK");
    return from !== undefined && blockNumber >= from;
  }
  return true;
}
```

`src/lib/primaryPool.ts`:
```ts
import { toUnits } from "./price";

export type PoolRank = {
  id: string;
  quoteId: string;
  tvlToken: bigint | undefined;
  tvlQuote: bigint | undefined;
  priceQuote: number;
  tokenDecimals: number;
  quoteDecimals: number;
  swapCount: number;
  createdAt: number;
};

/** Highest USD TVL, then swap count, then earliest pool (spec section 5). */
export function choosePrimaryPool(
  pools: PoolRank[],
  usdPrice: (quoteId: string) => number | undefined,
): string | undefined {
  if (pools.length === 0) return undefined;
  const scored = pools.map((p) => {
    const hasTvl = p.tvlToken !== undefined && p.tvlQuote !== undefined;
    const usd = usdPrice(p.quoteId) ?? 0;
    const tvlQuoteUnits = hasTvl
      ? toUnits(p.tvlQuote!, p.quoteDecimals) + toUnits(p.tvlToken!, p.tokenDecimals) * p.priceQuote
      : 0;
    return { p, tvlUsd: tvlQuoteUnits * usd };
  });
  scored.sort((a, b) =>
    b.tvlUsd - a.tvlUsd ||
    b.p.swapCount - a.p.swapCount ||
    a.p.createdAt - b.p.createdAt ||
    a.p.id.localeCompare(b.p.id),
  );
  return scored[0]!.p.id;
}
```

```ts
import JSBI from "jsbi";
import { TickMath, SqrtPriceMath } from "@uniswap/v3-sdk";

const toJ = (x: bigint) => JSBI.BigInt(x.toString());
const toB = (x: JSBI) => BigInt(x.toString());

/**
 * Raw token amounts a liquidity change moves into (+) or out of (-) a pool.
 * Same piecewise rule as Uniswap's Pool.modifyLiquidity: only the part of the
 * range around the current price holds both currencies. Price does not move
 * between swaps, so the pool's stored sqrtPriceX96 is the right price here.
 */
export function amountsForLiquidityDelta(
  sqrtPriceX96: bigint,
  tickLower: number,
  tickUpper: number,
  liquidityDelta: bigint,
): { amount0: bigint; amount1: bigint } {
  if (liquidityDelta === 0n) return { amount0: 0n, amount1: 0n };
  const adding = liquidityDelta > 0n;
  const liquidity = toJ(adding ? liquidityDelta : -liquidityDelta);
  const sqrtCurrent = toJ(sqrtPriceX96);
  const sqrtLower = TickMath.getSqrtRatioAtTick(tickLower);
  const sqrtUpper = TickMath.getSqrtRatioAtTick(tickUpper);
  const currentTick = TickMath.getTickAtSqrtRatio(sqrtCurrent);

  let amount0 = JSBI.BigInt(0);
  let amount1 = JSBI.BigInt(0);
  if (currentTick < tickLower) {
    amount0 = SqrtPriceMath.getAmount0Delta(sqrtLower, sqrtUpper, liquidity, adding);
  } else if (currentTick < tickUpper) {
    amount0 = SqrtPriceMath.getAmount0Delta(sqrtCurrent, sqrtUpper, liquidity, adding);
    amount1 = SqrtPriceMath.getAmount1Delta(sqrtLower, sqrtCurrent, liquidity, adding);
  } else {
    amount1 = SqrtPriceMath.getAmount1Delta(sqrtLower, sqrtUpper, liquidity, adding);
  }
  const sign = adding ? 1n : -1n;
  return { amount0: sign * toB(amount0), amount1: sign * toB(amount1) };
}
```

```ts
import { createEffect, S } from "envio";
import { createPublicClient, http, parseAbi, hexToString } from "viem";

const RPC_URL = process.env.ENVIO_RPC_URL;

const client = RPC_URL
  ? createPublicClient({ transport: http(RPC_URL, { batch: true }) })
  : undefined;

const ERC20 = parseAbi([
  "function symbol() view returns (string)",
  "function name() view returns (string)",
  "function decimals() view returns (uint8)",
]);
const ERC20_BYTES = parseAbi([
  "function symbol() view returns (bytes32)",
  "function name() view returns (bytes32)",
  "function decimals() view returns (uint8)",
]);

function fallback(address: string) {
  return { symbol: address.slice(0, 6), name: address, decimals: 18 };
}

// bytes32 names are padded with NUL bytes; drop them.
function stripNulls(s: string): string {
  return s.replace(/\u0000/g, "");
}

/** Name, symbol and decimals of an ERC-20, cached forever per address. */
export const getTokenMetadata = createEffect(
  {
    name: "getTokenMetadata",
    input: S.string,
    output: { symbol: S.string, name: S.string, decimals: S.number },
    cache: true,
    rateLimit: false,
  },
  async ({ input: address, context }) => {
    if (!client) return fallback(address);
    const addr = address as `0x${string}`;
    try {
      const [symbol, name, decimals] = await Promise.all([
        client.readContract({ address: addr, abi: ERC20, functionName: "symbol" }),
        client.readContract({ address: addr, abi: ERC20, functionName: "name" }),
        client.readContract({ address: addr, abi: ERC20, functionName: "decimals" }),
      ]);
      return { symbol, name, decimals: Number(decimals) };
    } catch {
      try {
        const [symbol, name, decimals] = await Promise.all([
          client.readContract({ address: addr, abi: ERC20_BYTES, functionName: "symbol" }),
          client.readContract({ address: addr, abi: ERC20_BYTES, functionName: "name" }),
          client.readContract({ address: addr, abi: ERC20_BYTES, functionName: "decimals" }),
        ]);
        return { symbol: stripNulls(hexToString(symbol)), name: stripNulls(hexToString(name)), decimals: Number(decimals) };
      } catch (err) {
        context.log.warn(`token metadata failed for ${address}: ${String(err)}`);
        return fallback(address);
      }
    }
  },
);
```

```ts
import { indexer, type Quote, type Token } from "envio";
import { quoteRow } from "../lib/quotes";
import { choosePrimaryPool } from "../lib/primaryPool";
import { USDG } from "../lib/constants";

type Handler = Parameters<typeof indexer.onEvent>[1];
export type HandlerContext = Parameters<Handler>[0]["context"];

/** Seeds the Quote row (and USDG's pinned price) the first time a quote is seen. */
export async function ensureQuote(context: HandlerContext, address: string, timestamp: number): Promise<Quote> {
  const id = address.toLowerCase();
  const existing = await context.Quote.get(id);
  if (existing) return existing;
  const row = quoteRow(id);
  if (!row) throw new Error(`ensureQuote called for non-quote ${address}`);
  const quote: Quote = { id, address, symbol: row.symbol, name: row.name, decimals: row.decimals, enabled: true };
  context.Quote.set(quote);
  if (id === USDG) {
    context.QuotePrice.set({ id, quote_id: id, usdPrice: 1, updatedAt: timestamp, aggregator: "pinned" });
  }
  return quote;
}

/** Token row that Transfer or Initialize can create first; metadata is filled by Initialize. */
export async function ensureToken(context: HandlerContext, address: string, timestamp: number): Promise<Token> {
  const id = address.toLowerCase();
  const existing = await context.Token.get(id);
  if (existing) return existing;
  const token: Token = {
    id, address, symbol: address.slice(0, 6), name: address, decimals: 18, metadataLoaded: false,
    totalSupply: 0n, holdersCount: 0, poolCount: 0, origin: "Other", primaryPool_id: undefined, createdAt: timestamp,
  };
  context.Token.set(token);
  return token;
}

/** Re-picks primaryPool (and the token's origin) from the token's pools. */
export async function recomputePrimaryPool(context: HandlerContext, tokenId: string): Promise<void> {
  const token = await context.Token.get(tokenId);
  if (!token) return;
  const pools = await context.Pool.getWhere({ token_id: { _eq: tokenId } });
  if (pools.length === 0) return;
  const prices = new Map<string, number | undefined>();
  const quoteDecimals = new Map<string, number>();
  for (const p of pools) {
    if (!prices.has(p.quote_id)) prices.set(p.quote_id, (await context.QuotePrice.get(p.quote_id))?.usdPrice);
    if (!quoteDecimals.has(p.quote_id)) quoteDecimals.set(p.quote_id, (await context.Quote.get(p.quote_id))?.decimals ?? 18);
  }
  const best = choosePrimaryPool(
    pools.map((p) => ({
      id: p.id, quoteId: p.quote_id, tvlToken: p.tvlToken, tvlQuote: p.tvlQuote, priceQuote: p.priceQuote,
      tokenDecimals: token.decimals, quoteDecimals: quoteDecimals.get(p.quote_id) ?? 18,
      swapCount: p.swapCount, createdAt: p.createdAt,
    })),
    (q) => prices.get(q),
  );
  const bestPool = pools.find((p) => p.id === best);
  if (bestPool && (token.primaryPool_id !== best || token.origin !== bestPool.origin)) {
    context.Token.set({ ...token, primaryPool_id: best, origin: bestPool.origin });
  }
}
```

```ts
import { indexer, type Pool, type Swap, type Candle } from "envio";
import { isQuote, QUOTE_ADDRESSES } from "../lib/quotes";
import { originFor } from "../lib/origin";
import { priceFromSqrtPriceX96, toUnits, absBig } from "../lib/price";
import { INTERVALS, bucketStart, candleId, candleWindowAllows } from "../lib/candles";
import { amountsForLiquidityDelta } from "../lib/liquidity";
import { getTokenMetadata } from "../effects/tokenMetadata";
import { ensureQuote, ensureToken, recomputePrimaryPool, type HandlerContext } from "./shared";

// HyperSync only sends Initialize logs where one of the indexed sides is a quote.
const INITIALIZE = {
  contract: "PoolManager",
  event: "Initialize",
  where: { params: [{ currency0: QUOTE_ADDRESSES }, { currency1: QUOTE_ADDRESSES }] },
} as const;

function classify(currency0: string, currency1: string) {
  const q0 = isQuote(currency0);
  const q1 = isQuote(currency1);
  if (q0 === q1) return null; // both quotes or neither: not a memestock pool
  return q1
    ? { token: currency0, quote: currency1, tokenIsCurrency0: true }
    : { token: currency1, quote: currency0, tokenIsCurrency0: false };
}

indexer.contractRegister(INITIALIZE, async ({ event, context }) => {
  const c = classify(event.params.currency0, event.params.currency1);
  if (c) context.chain.Token.add(c.token);
});

indexer.onEvent({ ...INITIALIZE, fields: { block: ["timestamp"] } }, async ({ event, context }) => {
  const c = classify(event.params.currency0, event.params.currency1);
  if (!c) return;
  const poolId = event.params.id.toLowerCase();
  const ts = event.block.timestamp;
  const quote = await ensureQuote(context, c.quote, ts);
  let token = await ensureToken(context, c.token, ts);
  if (!token.metadataLoaded) {
    const meta = await context.effect(getTokenMetadata, c.token);
    token = { ...token, symbol: meta.symbol, name: meta.name, decimals: meta.decimals, metadataLoaded: true };
  }
  const dec0 = c.tokenIsCurrency0 ? token.decimals : quote.decimals;
  const dec1 = c.tokenIsCurrency0 ? quote.decimals : token.decimals;
  const origin = originFor(poolId, event.params.hooks);
  const pool: Pool = {
    id: poolId,
    token_id: token.id,
    quote_id: quote.id,
    tokenIsCurrency0: c.tokenIsCurrency0,
    hooks: event.params.hooks.toLowerCase(),
    fee: Number(event.params.fee),
    tickSpacing: Number(event.params.tickSpacing),
    origin,
    createdAt: ts,
    priceQuote: priceFromSqrtPriceX96(event.params.sqrtPriceX96, dec0, dec1, c.tokenIsCurrency0),
    sqrtPriceX96: event.params.sqrtPriceX96,
    tvlToken: 0n,
    tvlQuote: 0n,
    swapCount: 0,
    lastSwapAt: undefined,
    volumeQuoteAllTime: 0,
  };
  context.Pool.set(pool);
  const poolCount = token.poolCount + 1;
  context.Token.set({
    ...token,
    poolCount,
    primaryPool_id: token.primaryPool_id ?? poolId,
    origin: token.primaryPool_id ? token.origin : origin,
  });
  if (poolCount > 1) await recomputePrimaryPool(context, token.id);
});

function updateCandles(
  context: HandlerContext,
  poolId: string,
  prevPrice: number,
  price: number,
  amountQuote: number,
  amountToken: number,
  timestamp: number,
  blockNumber: number,
): Promise<void>[] {
  return INTERVALS.filter((iv) => candleWindowAllows(iv, blockNumber)).map(async (interval) => {
    const t = bucketStart(timestamp, interval);
    const id = candleId(poolId, interval, t);
    const existing = await context.Candle.get(id);
    if (existing) {
      context.Candle.set({
        ...existing,
        high: Math.max(existing.high, price),
        low: Math.min(existing.low, price),
        close: price,
        volumeQuote: existing.volumeQuote + amountQuote,
        volumeToken: existing.volumeToken + amountToken,
        trades: existing.trades + 1,
      });
    } else {
      // Gapless: a bucket after a gap opens at the previous pool price.
      const candle: Candle = {
        id, pool_id: poolId, interval, t,
        open: prevPrice, high: Math.max(prevPrice, price), low: Math.min(prevPrice, price), close: price,
        volumeQuote: amountQuote, volumeToken: amountToken, trades: 1,
      };
      context.Candle.set(candle);
    }
  });
}

indexer.onEvent(
  { contract: "PoolManager", event: "Swap", fields: { block: ["timestamp"], transaction: ["hash", "from"] } },
  async ({ event, context }) => {
    const pool = await context.Pool.get(event.params.id.toLowerCase());
    if (!pool) return; // untracked pool
    const token = await context.Token.getOrThrow(pool.token_id);
    const quote = await context.Quote.getOrThrow(pool.quote_id);
    // Swap amounts are the swapper's deltas, emitted before afterSwap: they are pre-hook-fee.
    const tokenDelta = pool.tokenIsCurrency0 ? event.params.amount0 : event.params.amount1;
    const quoteDelta = pool.tokenIsCurrency0 ? event.params.amount1 : event.params.amount0;
    const amountToken = toUnits(absBig(tokenDelta), token.decimals);
    const amountQuote = toUnits(absBig(quoteDelta), quote.decimals);
    const dec0 = pool.tokenIsCurrency0 ? token.decimals : quote.decimals;
    const dec1 = pool.tokenIsCurrency0 ? quote.decimals : token.decimals;
    const price = priceFromSqrtPriceX96(event.params.sqrtPriceX96, dec0, dec1, pool.tokenIsCurrency0);
    const ts = event.block.timestamp;

    const swap: Swap = {
      id: `${event.block.number}_${event.logIndex}`,
      pool_id: pool.id,
      token_id: token.id,
      timestamp: ts,
      side: tokenDelta > 0n ? "buy" : "sell",
      amountToken,
      amountQuote,
      priceQuote: amountToken > 0 ? amountQuote / amountToken : price,
      wallet: event.transaction.from.toLowerCase(),
      txHash: event.transaction.hash,
    };
    context.Swap.set(swap);

    await Promise.all(updateCandles(context, pool.id, pool.priceQuote, price, amountQuote, amountToken, ts, event.block.number));

    context.Pool.set({
      ...pool,
      priceQuote: price,
      sqrtPriceX96: event.params.sqrtPriceX96,
      // The pool gains what the swapper pays and loses what the swapper receives.
      tvlToken: pool.tvlToken === undefined ? undefined : pool.tvlToken - tokenDelta,
      tvlQuote: pool.tvlQuote === undefined ? undefined : pool.tvlQuote - quoteDelta,
      swapCount: pool.swapCount + 1,
      lastSwapAt: ts,
      volumeQuoteAllTime: pool.volumeQuoteAllTime + amountQuote,
    });
    if (token.poolCount > 1) await recomputePrimaryPool(context, token.id);
  },
);

indexer.onEvent({ contract: "PoolManager", event: "ModifyLiquidity" }, async ({ event, context }) => {
  const pool = await context.Pool.get(event.params.id.toLowerCase());
  if (!pool || pool.tvlToken === undefined || pool.tvlQuote === undefined) return;
  const { amount0, amount1 } = amountsForLiquidityDelta(
    pool.sqrtPriceX96,
    Number(event.params.tickLower),
    Number(event.params.tickUpper),
    event.params.liquidityDelta,
  );
  const tokenAmt = pool.tokenIsCurrency0 ? amount0 : amount1;
  const quoteAmt = pool.tokenIsCurrency0 ? amount1 : amount0;
  context.Pool.set({ ...pool, tvlToken: pool.tvlToken + tokenAmt, tvlQuote: pool.tvlQuote + quoteAmt });
  const token = await context.Token.getOrThrow(pool.token_id);
  if (token.poolCount > 1) await recomputePrimaryPool(context, token.id);
});

indexer.onEvent({ contract: "PoolManager", event: "Donate" }, async ({ event, context }) => {
  const pool = await context.Pool.get(event.params.id.toLowerCase());
  if (!pool || pool.tvlToken === undefined || pool.tvlQuote === undefined) return;
  const tokenAmt = pool.tokenIsCurrency0 ? event.params.amount0 : event.params.amount1;
  const quoteAmt = pool.tokenIsCurrency0 ? event.params.amount1 : event.params.amount0;
  context.Pool.set({ ...pool, tvlToken: pool.tvlToken + tokenAmt, tvlQuote: pool.tvlQuote + quoteAmt });
});
```

```ts
import { indexer, type TokenBalance } from "envio";
import { ZERO_ADDRESS } from "../lib/constants";
import { ensureToken } from "./shared";

indexer.onEvent({ contract: "Token", event: "Transfer", fields: { block: ["timestamp"] } }, async ({ event, context }) => {
  const ts = event.block.timestamp;
  const token = await ensureToken(context, event.srcAddress, ts);
  const from = event.params.from.toLowerCase();
  const to = event.params.to.toLowerCase();
  const value = event.params.value;
  if (from === to) return; // self-transfer: nothing moves
  let holdersDelta = 0;
  let supplyDelta = 0n;

  if (from === ZERO_ADDRESS) {
    supplyDelta += value;
  } else {
    const id = `${token.id}-${from}`;
    const bal = await context.TokenBalance.get(id);
    const before = bal?.balance ?? 0n;
    const after = before - value;
    if (before > 0n && after <= 0n) holdersDelta -= 1;
    const row: TokenBalance = { id, token_id: token.id, holder: from, balance: after, updatedAt: ts };
    context.TokenBalance.set(row);
  }

  if (to === ZERO_ADDRESS) {
    supplyDelta -= value;
  } else {
    const id = `${token.id}-${to}`;
    const bal = await context.TokenBalance.get(id);
    const before = bal?.balance ?? 0n;
    const after = before + value;
    if (before <= 0n && after > 0n) holdersDelta += 1;
    const row: TokenBalance = { id, token_id: token.id, holder: to, balance: after, updatedAt: ts };
    context.TokenBalance.set(row);
  }

  if (holdersDelta !== 0 || supplyDelta !== 0n) {
    context.Token.set({
      ...token,
      holdersCount: token.holdersCount + holdersDelta,
      totalSupply: token.totalSupply + supplyDelta,
    });
  }
});
```

```ts
import { indexer, type QuoteFeed } from "envio";
import { quoteForFeed } from "../lib/quotes";
import { ensureQuote } from "./shared";

indexer.contractRegister({ contract: "ChainlinkProxy", event: "AggregatorConfirmed" }, async ({ event, context }) => {
  context.chain.ChainlinkAggregator.add(event.params.latest);
});

indexer.onEvent(
  { contract: "ChainlinkProxy", event: "AggregatorConfirmed", fields: { block: ["timestamp"] } },
  async ({ event, context }) => {
    const row = quoteForFeed(event.srcAddress);
    if (!row || row.chainlinkDecimals === null) return;
    const quote = await ensureQuote(context, row.address, event.block.timestamp);
    const feed: QuoteFeed = {
      id: event.params.latest.toLowerCase(),
      quote_id: quote.id,
      decimals: row.chainlinkDecimals,
      proxy: event.srcAddress.toLowerCase(),
    };
    context.QuoteFeed.set(feed);
  },
);

indexer.onEvent(
  { contract: "ChainlinkAggregator", event: "AnswerUpdated", fields: { block: ["timestamp"] } },
  async ({ event, context }) => {
    const aggregator = event.srcAddress.toLowerCase();
    let feed = await context.QuoteFeed.get(aggregator);
    if (!feed) {
      // Static aggregator from config/quotes.json, seen before any AggregatorConfirmed.
      const row = quoteForFeed(aggregator);
      if (!row || row.chainlinkDecimals === null || !row.chainlinkProxy) return;
      const quote = await ensureQuote(context, row.address, event.block.timestamp);
      feed = { id: aggregator, quote_id: quote.id, decimals: row.chainlinkDecimals, proxy: row.chainlinkProxy };
      context.QuoteFeed.set(feed);
    }
    if (event.params.current <= 0n) return; // Chainlink reports errors as non-positive answers
    context.QuotePrice.set({
      id: feed.quote_id,
      quote_id: feed.quote_id,
      usdPrice: Number(event.params.current) / 10 ** feed.decimals,
      updatedAt: Number(event.params.updatedAt),
      aggregator,
    });
  },
);
```
